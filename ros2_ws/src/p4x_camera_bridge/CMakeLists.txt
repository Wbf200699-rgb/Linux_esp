cmake_minimum_required(VERSION 3.16)
project(p4x_camera_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(CURL REQUIRED)
add_executable(camera_bridge src/camera_bridge.cpp)
target_compile_features(camera_bridge PRIVATE cxx_std_17)
target_compile_options(camera_bridge PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(camera_bridge rclcpp sensor_msgs cv_bridge)
target_link_libraries(camera_bridge CURL::libcurl opencv_core opencv_imgcodecs)
install(TARGETS camera_bridge DESTINATION lib/p4x_camera_bridge)
install(DIRECTORY launch DESTINATION share/p4x_camera_bridge)
ament_package()
